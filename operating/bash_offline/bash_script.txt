#22A1


#Recursively classify files by their dominant keyword,
#append each file’s total keyword score, move it into the #corresponding category,
#and number files by ascending size within each category.

if [ "$#" -ne 2 ]; then
    echo "Usage: $0 <input> <output>"
    exit 1
fi

INPUT="$1"
OUTPUT="$2"

if [ ! -d "$INPUT" ]; then
    echo "Error: '$INPUT' is not a directory."
    exit 1
fi

mkdir -p "$OUTPUT/Stark"
mkdir -p "$OUTPUT/Targaryen"
mkdir -p "$OUTPUT/Royal"


while IFS= read -r -d '' file; do
    filename=$(basename "$file")
    winter_count=$(grep -ic winter "$file")
    dragon_count=$(grep -ic dragon "$file")
    throne_count=$(grep -ic throne "$file")
    total_count=$((winter_count+dragon_count+throne_count))
    if [ $winter_count -gt "$dragon_count" ] && [ "$winter_count" -gt "$throne_count" ]; then
        category="Stark"
    elif [ "$winter_count" -lt "$dragon_count" ] && [ "$dragon_count" -gt "$throne_count" ]; then
        category="Targaryen"
    else
        category="Royal"
    fi

    cp "$file" "$OUTPUT/$category/${filename}_${total_count}"

done < <(find "$INPUT" -type f -print0)

for category in Stark Targaryen Royal; do
    count=0
    while IFS= read -r -d '' entry; do
    file="${entry#* }"
    directory=$(dirname "$file")
    filename=$(basename "$file")
    mv "$file" "$directory/${count}_$filename"
    ((count++))

    done < <(find "$OUTPUT/$category" -type f -printf '%s %p\0' | sort -z )
done

#Scan a media directory, extract artist and title information from supported media filenames, organize
#them by artist, and generate a sorted catalog report.

# Check command-line arguments
if [ "$#" -ne 2 ]; then
    echo "Usage: $0 media_directory catalog_file"
    exit 1
fi

MEDIA_DIR="$1"
OUTPUT_FILE="$2"

# Check whether the media directory exists
if [ ! -d "$MEDIA_DIR" ]; then
    echo "Error: '$MEDIA_DIR' is not a valid directory."
    exit 1
fi

# Associative array:
# artist name -> newline-separated list of titles
declare -A catalog

# Read supported media files safely, including filenames with spaces
while IFS= read -r -d '' file; do
    filename=$(basename "$file")

    # Remove the file extension
    name="${filename%.*}"

    artist="Unknown"
    title="Unknown"

    # Pattern: Title (Year) - Artist
    if [[ "$name" =~ ^(.+)[[:space:]]+\([0-9]{4}\)[[:space:]]+-[[:space:]]+(.+)$ ]]; then
        title="${BASH_REMATCH[1]}"
        artist="${BASH_REMATCH[2]}"

    # Pattern: Artist - Title
    elif [[ "$name" =~ ^(.+)[[:space:]]+-[[:space:]]+(.+)$ ]]; then
        artist="${BASH_REMATCH[1]}"
        title="${BASH_REMATCH[2]}"
    fi

    # Store the title under its artist
    catalog["$artist"]+="$title"$'\n'

done < <(
    find "$MEDIA_DIR" -maxdepth 1 -type f \
        \( -iname "*.mp3" -o -iname "*.flac" \
           -o -iname "*.mp4" -o -iname "*.mkv" \) \
        -print0
)

# Empty/create the output file
> "$OUTPUT_FILE"

# Sort artists alphabetically
while IFS= read -r artist; do
    echo "$artist" >> "$OUTPUT_FILE"

    # Sort this artist's titles alphabetically
    printf '%s' "${catalog[$artist]}" |
        sort >> "$OUTPUT_FILE"

done < <(printf '%s\n' "${!catalog[@]}" | sort)

#Create a Bash script that scans HTML files for broken internal `<a href="...">` links,
#skips unchanged files using a timestamp cache, and generates a summary report.

# Usage:
# ./script.sh <directory> [cache_file_path]

if [ "$#" -lt 1 ] || [ "$#" -gt 2 ]; then
    echo "Usage: $0 <directory> [cache_file_path]"
    exit 1
fi

DIRECTORY="${1%/}"
CACHE_FILE="${2:-cache.txt}"
REPORT_FILE="report.txt"

if [ ! -d "$DIRECTORY" ]; then
    echo "Error: Directory does not exist."
    exit 1
fi

SCAN_TIME=$(date '+%Y-%m-%d %H:%M:%S')
CURRENT_TIMESTAMP=$(date +%s)

total_html_files=0
files_checked=0
cache_hits=0
cache_misses=0
total_internal_links=0
total_broken_links=0

declare -A cache
declare -A new_cache
declare -A broken_counts

# Read the existing cache file.
if [ -f "$CACHE_FILE" ]; then
    while IFS= read -r line; do
        # Skip the heading and empty lines.
        if [ -z "$line" ] || [[ "$line" == "Cache File - Last updated:"* ]]; then
            continue
        fi

        file_path="${line%:*}"
        last_scan_time="${line##*:}"

        cache["$file_path"]="$last_scan_time"
    done < "$CACHE_FILE"
fi

# Find every HTML file recursively.
while IFS= read -r html_file; do
    ((total_html_files++))

    modification_time=$(stat --format=%Y "$html_file")
    last_scan_time="${cache[$html_file]}"

    # Check only new files or files modified since the last scan.
    if [ -z "$last_scan_time" ] || [ "$modification_time" -gt "$last_scan_time" ]; then
        ((files_checked++))
        ((cache_misses++))

        broken_count=0

        # Extract links only from:
        # <a href="...">
        while IFS= read -r link; do
            [ -z "$link" ] && continue

            ((total_internal_links++))

            # Relative links are resolved from the HTML file's directory.
            if [[ "$link" == /* ]]; then
                target="$DIRECTORY/${link#/}"
            else
                target="$(dirname "$html_file")/$link"
            fi

            if [ ! -e "$target" ]; then
                ((broken_count++))
                ((total_broken_links++))
            fi

        done < <(
            grep -o '<a href="[^"]*">' "$html_file" |
            sed 's/<a href="//; s/">//'
        )

        broken_counts["$html_file"]="$broken_count"

        # Store the time when this file was scanned.
        new_cache["$html_file"]="$CURRENT_TIMESTAMP"
    else
        ((cache_hits++))

        # Preserve the previous last-scanned timestamp.
        new_cache["$html_file"]="$last_scan_time"
    fi

done < <(find "$DIRECTORY" -type f -iname "*.html")

# Generate report.
{
    echo "Scan time: $SCAN_TIME"
    echo "Directory: $DIRECTORY"
    echo
    echo "BROKEN LINK COUNTS:"

    for file in "${!broken_counts[@]}"; do
        echo "$file: ${broken_counts[$file]}"
    done

    echo
    echo "STATISTICS:"
    echo "Total HTML files scanned: $total_html_files"
    echo "Files checked: $files_checked ($cache_hits from cache)"
    echo "Total internal links: $total_internal_links"
    echo "Broken links: $total_broken_links"
    echo
    echo "CACHE SUMMARY:"
    echo "Cache hits: $cache_hits"
    echo "Cache misses: $cache_misses"
} > "$REPORT_FILE"

# Update cache file.
{
    echo "Cache File - Last updated: $SCAN_TIME"

    for file in "${!new_cache[@]}"; do
        echo "$file:${new_cache[$file]}"
    done
} > "$CACHE_FILE"

echo "Scan completed."
echo "Report generated in $REPORT_FILE"
echo "Cache updated in $CACHE_FILE"

# Photo Album Curator
# Organizes photos based on capture time.

PHOTO_DIR="photos"

mkdir -p "$PHOTO_DIR/morning"
mkdir -p "$PHOTO_DIR/afternoon"
mkdir -p "$PHOTO_DIR/evening"

morning_count=0
afternoon_count=0
evening_count=0


# Scan all photos matching IMG_YYYYMMDD_HHMMSS.jpg
for file in "$PHOTO_DIR"/IMG_*.jpg; do

    # Skip if no matching files exist
    [ -e "$file" ] || continue

    filename=$(basename "$file")


    # Extract time part
    # Example:
    # IMG_20250501_074512.jpg
    # time_part = 074512

    time_part="${filename##*_}"
    time_part="${time_part%.jpg}"


    # Extract hour
    # 074512 -> 07
    hour="${time_part:0:2}"

    # Convert to decimal (avoid octal issue with 08, 09)
    hour=$((10#$hour))


    # Categorize photo

    if [ "$hour" -ge 0 ] && [ "$hour" -le 11 ]; then

        category="morning"
        ((morning_count++))

    elif [ "$hour" -ge 12 ] && [ "$hour" -le 17 ]; then

        category="afternoon"
        ((afternoon_count++))

    else

        category="evening"
        ((evening_count++))

    fi


    # Rename and move
    # Example:
    # IMG_20250501_074512.jpg
    # ->
    # morning_IMG_20250501_074512.jpg

    mv "$file" "$PHOTO_DIR/$category/${category}_$filename"

done


# Generate counts.txt

{
    echo "morning: $morning_count"
    echo "afternoon: $afternoon_count"
    echo "evening: $evening_count"
} > "$PHOTO_DIR/counts.txt"


echo "Photo organization completed."

#Identify and remove duplicate files recursively from a directory by filtering files by size,
#comparing MD5 hashes, keeping one copy, and generating a storage-saving report.

# File Deduplicator
#
# Usage:
# ./deduplicator.sh <target_directory>

if [ "$#" -ne 1 ]; then
    echo "Usage: $0 <target_directory>"
    exit 1
fi

TARGET_DIR="$1"
REPORT_FILE="report.txt"

if [ ! -d "$TARGET_DIR" ]; then
    echo "Error: Directory does not exist."
    exit 1
fi


total_files=0
duplicates_found=0
space_saved=0


# Associative array:
# size -> list of files having that size

declare -A size_groups


# --------------------------------------------------
# Step 1:
# Scan all files recursively and group by size
# --------------------------------------------------

while IFS= read -r -d '' file; do

    ((total_files++))

    size=$(stat --format=%s "$file")

    if [ -z "${size_groups[$size]}" ]; then
        size_groups[$size]="$file"
    else
        size_groups[$size]+=$'\n'"$file"
    fi

done < <(find "$TARGET_DIR" -type f -print0)



# --------------------------------------------------
# Step 2:
# For files having same size, compare MD5 hash
# --------------------------------------------------

for size in "${!size_groups[@]}"; do

    # Get files with same size
    mapfile -t files <<< "${size_groups[$size]}"

    # If only one file exists, no duplicate possible
    if [ "${#files[@]}" -le 1 ]; then
        continue
    fi


    # Hash groups:
    # md5 -> list of files

    declare -A hash_groups


    for file in "${files[@]}"; do

        hash=$(md5sum "$file" | cut -d ' ' -f 1)

        if [ -z "${hash_groups[$hash]}" ]; then
            hash_groups[$hash]="$file"
        else
            hash_groups[$hash]+=$'\n'"$file"
        fi

    done



    # --------------------------------------------------
    # Step 3:
    # Delete duplicate files from same hash group
    # --------------------------------------------------

    for hash in "${!hash_groups[@]}"; do

        mapfile -t duplicate_files <<< "${hash_groups[$hash]}"

        if [ "${#duplicate_files[@]}" -le 1 ]; then
            continue
        fi


        # Keep first file, remove remaining files

        for ((i=1; i<${#duplicate_files[@]}; i++)); do

            duplicate_file="${duplicate_files[$i]}"

            file_size=$(stat --format=%s "$duplicate_file")

            rm "$duplicate_file"

            ((duplicates_found++))
            ((space_saved+=file_size))

        done

    done


    # Clear hash array before next size group
    unset hash_groups

done



# --------------------------------------------------
# Generate report
# --------------------------------------------------

{
    echo "Total files scanned: $total_files"
    echo "Duplicates found: $duplicates_found"
    echo "Space saved: $space_saved bytes"
} > "$REPORT_FILE"


echo "Deduplication completed."
echo "Report generated: $REPORT_FILE"

#Traverse city directories, copy and rename all part files into a central `blueprints/` folder,
#then count even-numbered parts by category and generate a sorted inventory summary.


# Operation Golden Blueprint
#
# Usage:
# ./script.sh <directory>

if [ "$#" -ne 1 ]; then
    echo "Usage: $0 <directory>"
    exit 1
fi

BASE_DIR="$1"
BLUEPRINT_DIR="$BASE_DIR/blueprints"
SUMMARY_FILE="$BASE_DIR/inventory_summary.txt"

if [ ! -d "$BASE_DIR" ]; then
    echo "Error: Directory does not exist."
    exit 1
fi

mkdir -p "$BLUEPRINT_DIR"

declare -A category_count

# Traverse every city directory
for city_dir in "$BASE_DIR"/*; do

    [ -d "$city_dir" ] || continue

    city=$(basename "$city_dir")

    # Skip the output directory
    [ "$city" = "blueprints" ] && continue

    # Process files such as Part_02_gear.dat
    for file in "$city_dir"/Part_??_*.dat; do

        [ -e "$file" ] || continue

        filename=$(basename "$file")

        # Remove "Part_" from the beginning
        remaining="${filename#Part_}"

        # Extract part number
        # 02_gear.dat -> 02
        part_number="${remaining%%_*}"

        # Remove part number and underscore
        # 02_gear.dat -> gear.dat
        category="${remaining#*_}"

        # Remove .dat
        # gear.dat -> gear
        category="${category%.dat}"

        # Copy with city prefix
        cp "$file" "$BLUEPRINT_DIR/${city}_$filename"

        # Count only even-numbered parts
        if (( 10#$part_number % 2 == 0 )); then
            ((category_count["$category"]++))
        fi

    done

done

# Generate sorted inventory summary
{
    for category in "${!category_count[@]}"; do
        echo "$category: ${category_count[$category]}"
    done
} | sort > "$SUMMARY_FILE"

echo "Blueprint organization completed."
echo "Summary generated at $SUMMARY_FILE"

#Recursively reorganize website files into a deployment-ready folder structure and
#update all HTML links to match the new locations.


# Website Deployment Packager
#
# Usage:
#   ./script.sh <website_directory>

if [ "$#" -ne 1 ]; then
    echo "Usage: $0 <website_directory>"
    exit 1
fi

WEBSITE_DIR="${1%/}"

if [ ! -d "$WEBSITE_DIR" ]; then
    echo "Error: '$WEBSITE_DIR' is not a valid directory."
    exit 1
fi

# Create deploy/ beside the input website directory
PARENT_DIR=$(dirname "$WEBSITE_DIR")
DEPLOY_DIR="$PARENT_DIR/deploy"

# Remove the previous deploy directory and recreate it
rm -rf "$DEPLOY_DIR"

mkdir -p "$DEPLOY_DIR"
mkdir -p "$DEPLOY_DIR/assets/css"
mkdir -p "$DEPLOY_DIR/assets/js"
mkdir -p "$DEPLOY_DIR/assets/images"

# --------------------------------------------------
# Copy HTML files to the root of deploy/
# --------------------------------------------------

while IFS= read -r -d '' file; do
    filename=$(basename "$file")
    cp "$file" "$DEPLOY_DIR/$filename"
done < <(
    find "$WEBSITE_DIR" -type f \
        \( -iname "*.html" -o -iname "*.htm" \) \
        -print0
)

# --------------------------------------------------
# Copy CSS files to deploy/assets/css/
# --------------------------------------------------

while IFS= read -r -d '' file; do
    filename=$(basename "$file")
    cp "$file" "$DEPLOY_DIR/assets/css/$filename"
done < <(
    find "$WEBSITE_DIR" -type f -iname "*.css" -print0
)

# --------------------------------------------------
# Copy JavaScript files to deploy/assets/js/
# --------------------------------------------------

while IFS= read -r -d '' file; do
    filename=$(basename "$file")
    cp "$file" "$DEPLOY_DIR/assets/js/$filename"
done < <(
    find "$WEBSITE_DIR" -type f -iname "*.js" -print0
)

# --------------------------------------------------
# Copy image files to deploy/assets/images/
# --------------------------------------------------

while IFS= read -r -d '' file; do
    filename=$(basename "$file")
    cp "$file" "$DEPLOY_DIR/assets/images/$filename"
done < <(
    find "$WEBSITE_DIR" -type f \
        \( -iname "*.jpg" -o \
           -iname "*.png" -o \
           -iname "*.gif" -o \
           -iname "*.svg" \) \
        -print0
)

# --------------------------------------------------
# Update links in all copied HTML files
# --------------------------------------------------

while IFS= read -r -d '' html_file; do

    sed -i -E \
        -e 's#(href|src)="[^"]*/([^/"]+\.css)"#\1="assets/css/\2"#g' \
        -e 's#(href|src)="([^/"]+\.css)"#\1="assets/css/\2"#g' \
        -e 's#(href|src)="[^"]*/([^/"]+\.js)"#\1="assets/js/\2"#g' \
        -e 's#(href|src)="([^/"]+\.js)"#\1="assets/js/\2"#g' \
        -e 's#(href|src)="[^"]*/([^/"]+\.(jpg|png|gif|svg))"#\1="assets/images/\2"#g' \
        -e 's#(href|src)="([^/"]+\.(jpg|png|gif|svg))"#\1="assets/images/\2"#g' \
        -e 's#href="[^"]*/([^/"]+\.(html|htm))"#href="\1"#g' \
        "$html_file"

done < <(
    find "$DEPLOY_DIR" -maxdepth 1 -type f \
        \( -iname "*.html" -o -iname "*.htm" \) \
        -print0
)

echo "Website packaging completed."
echo "Deployment directory: $DEPLOY_DIR"
